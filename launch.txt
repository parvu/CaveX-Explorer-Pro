# 2. Process hygiene — ALWAYS check before AND after launching
# ============================================================
# This environment accumulates orphaned processes and stale FastDDS shared-memory
# segments extremely easily, especially across repeated launch/kill cycles -- this

# project. Prefer FEWER, more patient launch/kill cycles over rapid iteration.
ps aux | grep -iE "gz sim|ardurover|ardusub|micro_ros_agent|MicroXRCEAgent|rtabmap|icp_odometry|static_transform|robot_state_publisher|cmd_vel_to_ardu|track_cmd_vel|track_retract|parameter_bridge|nav2|controller_server|planner_server|bt_navigator|behavior_server|lifecycle_manager|smoother_server|route_server|velocity_smoother|waypoint_follower|collision_monitor|docking_server|explore_lite|explore_node|ate_evaluator|ground_truth_odom|web_telemetry_bridge|slam_pose_publisher|ros2 topic pub|ros2 launch|ros2 run|gz topic|mavproxy|sitl_dds_udp"

sleep 3

# --- 5a. NORMAL path: /cmd_vel -> cmd_vel_to_ardupilot.py -> ArduPilot GUIDED+armed
# -> /ap/twist/filtered -> track_cmd_vel_bridge.py -> gz_bridge -> track drive.
# ArduPilot needs a few seconds for EKF/prearm checks to settle before arming
# succeeds (first attempt may report result=False — normal, retries automatically).
# IMPORTANT: use a single persistent `-r N` publisher, NOT a shell loop spawning a
# new `ros2 topic pub --once` process every iteration -- each spawn needs real DDS
# discovery time that can exceed a short sleep, so a spawn-loop effectively never
# actually delivers commands (confirmed the hard way).
ros2 topic pub -r 5 /cmd_vel geometry_msgs/msg/Twist "{linear: {x: 0.4}, angular: {z: 0.0}}" &
DRIVEPID=$!
sleep 15
kill -9 $DRIVEPID

gz topic -t /world/cavex_world/pose/info -e -n 1 | grep -A4 'name: "cavex_tracked_blueboat"'
# Expect real x movement from the spawn point (-35).

# Confirm ArduPilot state directly if needed:
ros2 service call /ap/mode_switch ardupilot_msgs/srv/ModeSwitch "{mode: 15}"   # 15 = GUIDED (Rover)
ros2 service call /ap/arm_motors ardupilot_msgs/srv/ArmMotors "{arm: true}"

# --- 5b. gz-transport workaround (bypasses ArduPilot entirely — still useful for
# quick motion tests that don't need ArduPilot in the loop):
( while true; do gz topic -t /model/cavex_tracked_blueboat/cmd_vel -m gz.msgs.Twist -p "linear: {x: 0.5}"; sleep 0.1; done ) &
DRIVEPID=$!
sleep 15
kill -9 $DRIVEPID
gz topic -t /world/cavex_world/pose/info -e -n 1 | grep -A4 'name: "cavex_tracked_blueboat"'

# Track retraction check (independent of driving):

sleep 3
ros2 topic echo /joint_states --once | grep -A2 "track_retract"

# ============================================================
# 8. Water region + BlueROV2 (standalone verification — real, live-tested)
# ============================================================
# UPDATED (re-derived alongside the floor-collision fix, section 0): real chamber
# floor z=5.9 (same CAVE_FLOOR_Z as cave_floor_patch), water surface at z=7.9 (a
# real 2m partial flood), region x [15,65] y [-10,10], centered (40, 0).
#
# The entity name MUST be exactly "bluerov2" (not "bluerov2_test") -- the Buoyancy

# <model name>, and silently no-ops on a mismatch (confirmed the hard way).
gz service -s /world/cavex_world/create --reqtype gz.msgs.EntityFactory --reptype gz.msgs.Boolean --timeout 3000 \
  --req 'sdf_filename: "src/cavex_tracked_vehicle/models/bluerov2/model.sdf", name: "bluerov2", pose: {position: {x: 40, y: 0, z: 6.9}}'
sleep 8
gz topic -t /world/cavex_world/pose/info -e -n 1 | grep -A5 'name: "bluerov2"'
# Expected: settles and stays roughly mid-column (z between 5.9 and 7.9), doesn't
# sink to the floor or breach the surface. Live-confirmed: spawned at z=6.9, settled
# z~7.2-7.5, stable-ish (drifts slowly toward the surface without active depth hold,
# real buoyancy behavior, not a bug).
